B1. Чем отличается адаптер от репозитория в контексте сервисного слоя?
Репозиторий (Repository):
Абстракция над хранилищем данных (БД, файлы, кеш);
Предоставляет CRUD-операции и поиск;
Работает с доменными объектами;
Скрывает детали реализации хранения

Адаптер (Adapter):
Преобразует один интерфейс в другой;
Адаптирует внешние системы к ожидаемому интерфейсу;
Может выполнять трансформацию данных, протоколов;
Часто используется для интеграции с внешними API

B2. Как реализовать транзакционность, если репозитории используют разные хранилища (SQL + NoSQL)?

Решение: Паттерн "Сага" (Saga Pattern)

class DistributedTransactionManager:
    def __init__(self):
        self._participants = []
        self._commit_log = []
    
    def register(self, participant):
        self._participants.append(participant)
    
    def execute(self, operations):
        try:
            # Фаза 1: Подготовка (Prepare)
            for participant in self._participants:
                participant.prepare()
                self._commit_log.append(participant)
            
            # Фаза 2: Фиксация (Commit)
            for participant in self._participants:
                participant.commit()
                
        except Exception as e:
            # Откат (Rollback)
            for participant in reversed(self._commit_log):
                participant.rollback()
            raise

B3. Как внедрить кеширование в сервис, не нарушая принцип единственной ответственности?

Решение: Паттерн "Декоратор" (Decorator Pattern)

from functools import wraps
import json
from typing import Any

class CacheDecorator:
    """Декоратор для добавления кеширования без изменения сервиса"""
    
    def __init__(self, cache_client, ttl=300):
        self._cache = cache_client
        self._ttl = ttl
    
    def cache_result(self, key_prefix: str):
        def decorator(func):
            @wraps(func)
            def wrapper(*args, **kwargs):
                # Формирование ключа
                cache_key = f"{key_prefix}:{self._make_key(*args, **kwargs)}"
                
                # Попытка получить из кеша
                cached = self._cache.get(cache_key)
                if cached is not None:
                    return json.loads(cached)
                
                # Вызов оригинального метода
                result = func(*args, **kwargs)
                
                # Сохранение в кеш
                self._cache.setex(cache_key, self._ttl, json.dumps(result))
                return result
            return wrapper
        return decorator
    
    def _make_key(self, *args, **kwargs):
        return json.dumps([args, kwargs], sort_keys=True)

class AutoparkServiceWithCache:
    def __init__(self, service: AutoparkService, cache_decorator: CacheDecorator):
        self._service = service
        self._cache = cache_decorator
    
    @CacheDecorator.cache_result("vehicle")
    def get_vehicle(self, vehicle_id: int) -> VehicleResponse:
        # Делегирование оригинальному сервису
        return self._service.get_vehicle(vehicle_id)

B7. В чём разница между валидацией на уровне DTO (схем) и валидацией в сервисе?

Аспект			|DTO валидация					|Сервисная валидация
----------------------------------------------------------------------------------------------------------------------
Что проверяет		|Структуру данных, типы, форматы		|Бизнес-правила, состояние системы
Зависимость		|Только от данных				|От состояния БД, других сервисов
Когда происходит	|При создании DTO				|В процессе выполнения бизнес-операции
Примеры			|Длина строки, диапазон чисел, формат email	|Проверка существования, статусов, доступности
Ошибки			|ValidationError				|DomainError, BusinessRuleViolation

Структурированное логирование с контекстом

import logging
from logging.config import dictConfig
import json
from datetime import datetime
from contextvars import ContextVar

# Контекстные переменные для передачи данных в логи
user_id_context: ContextVar[str] = ContextVar('user_id', default='anonymous')
request_id_context: ContextVar[str] = ContextVar('request_id', default='unknown')

class ContextualJSONFormatter(logging.Formatter):
    def format(self, record):
        log_data = {
            'timestamp': datetime.utcnow().isoformat(),
            'level': record.levelname,
            'logger': record.name,
            'message': record.getMessage(),
            'user_id': user_id_context.get(),
            'request_id': request_id_context.get(),
        }
        
        # Добавление дополнительных полей
        if hasattr(record, 'vehicle_id'):
            log_data['vehicle_id'] = record.vehicle_id
        if hasattr(record, 'trip_id'):
            log_data['trip_id'] = record.trip_id
        if hasattr(record, 'error_details'):
            log_data['error_details'] = record.error_details
        
        # Добавление исключения
        if record.exc_info:
            log_data['exception'] = self.formatException(record.exc_info)
        
        return json.dumps(log_data) + '\n'

# Настройка логирования
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

B9. Как организовать «мягкое удаление» записей через сервисный слой?

from datetime import datetime
from typing import Optional
from pydantic import BaseModel

# Модель с полями для мягкого удаления
class SoftDeleteMixin(BaseModel):
    is_deleted: bool = False
    deleted_at: Optional[datetime] = None

class Vehicle(SoftDeleteMixin):
    id: int
    license_plate: str
    brand: str
    model: str

# Репозиторий с поддержкой мягкого удаления
class VehicleRepository(ABC):
    @abstractmethod
    def soft_delete(self, vehicle_id: int) -> bool:
        """Мягкое удаление записи"""
        pass
    
    @abstractmethod
    def restore(self, vehicle_id: int) -> bool:
        """Восстановление записи"""
        pass
    
    @abstractmethod
    def get_all(self, include_deleted: bool = False) -> List[VehicleResponse]:
        """Получение всех записей с опцией включения удалённых"""
        pass

# Реализация In-Memory
class InMemoryVehicleRepository(VehicleRepository):
    def __init__(self):
        self._vehicles: Dict[int, Dict[str, Any]] = {}
        self._next_id = 1
    
    def soft_delete(self, vehicle_id: int) -> bool:
        if vehicle_id not in self._vehicles:
            return False
        
        self._vehicles[vehicle_id]['is_deleted'] = True
        self._vehicles[vehicle_id]['deleted_at'] = datetime.now()
        return True
    
    def restore(self, vehicle_id: int) -> bool:
        if vehicle_id not in self._vehicles:
            return False
        
        self._vehicles[vehicle_id]['is_deleted'] = False
        self._vehicles[vehicle_id]['deleted_at'] = None
        return True
    
    def get_all(self, include_deleted: bool = False) -> List[VehicleResponse]:
        items = list(self._vehicles.values())
        if not include_deleted:
            items = [v for v in items if not v.get('is_deleted', False)]
        return [self._to_response(item) for item in items]

# Сервисный слой
class AutoparkService:
    def delete_vehicle(self, vehicle_id: int, permanent: bool = False) -> bool:
        """Удаление ТС с опцией мягкого или жесткого удаления"""
        
        if not self._vehicles.exists(vehicle_id):
            raise VehicleNotFoundError(vehicle_id)
        
        # Проверка активных поездок
        trips = self._trips.get_by_vehicle(vehicle_id)
        active_trips = [t for t in trips if not t.is_completed]
        if active_trips:
            raise BusinessRuleViolation(
                f"Cannot delete vehicle with {len(active_trips)} active trips"
            )
        
        if permanent:
            # Жесткое удаление
            logger.warning(f"Permanently deleting vehicle {vehicle_id}")
            return self._vehicles.delete(vehicle_id)
        else:
            # Мягкое удаление
            logger.info(f"Soft deleting vehicle {vehicle_id}")
            return self._vehicles.soft_delete(vehicle_id)
    
    def restore_vehicle(self, vehicle_id: int) -> VehicleResponse:
        """Восстановление мягко удалённого ТС"""
        
        if not self._vehicles.exists(vehicle_id):
            raise VehicleNotFoundError(vehicle_id)
        
        # Проверка, что ТС действительно удалено
        vehicle = self._vehicles.get_by_id(vehicle_id)
        if not vehicle.is_deleted:
            raise BusinessRuleViolation(f"Vehicle {vehicle_id} is not deleted")
        
        self._vehicles.restore(vehicle_id)
        logger.info(f"Vehicle {vehicle_id} restored")
        return self._vehicles.get_by_id(vehicle_id)
    
    def get_vehicles(self, include_deleted: bool = False) -> List[VehicleResponse]:
        """Получение списка ТС с опцией включения удалённых"""
        return self._vehicles.get_all(include_deleted)

B10. Приведите пример пагинации в репозитории и использования в сервисе.

from typing import Optional, List, Generic, TypeVar
from pydantic import BaseModel

# Модель для пагинации
class PaginationParams(BaseModel):
    page: int = 1
    size: int = 20
    sort_by: Optional[str] = None
    sort_desc: bool = False

class Page(BaseModel, Generic[T]):
    items: List[T]
    page: int
    size: int
    total: int
    total_pages: int
    
    @property
    def has_next(self) -> bool:
        return self.page < self.total_pages
    
    @property
    def has_previous(self) -> bool:
        return self.page > 1

# Интерфейс репозитория с пагинацией
class VehicleRepository(ABC):
    @abstractmethod
    def get_all_paginated(self, params: PaginationParams) -> Page[VehicleResponse]:
        """Получение ТС с пагинацией"""
        pass
    
    @abstractmethod
    def find_by_status_paginated(self, status: str, params: PaginationParams) -> Page[VehicleResponse]:
        """Поиск по статусу с пагинацией"""
        pass

# Реализация In-Memory с пагинацией
class InMemoryVehicleRepository(VehicleRepository):
    def get_all_paginated(self, params: PaginationParams) -> Page[VehicleResponse]:
        items = list(self._vehicles.values())
        
        # Сортировка
        if params.sort_by and params.sort_by in items[0].keys():
            items.sort(
                key=lambda x: x[params.sort_by],
                reverse=params.sort_desc
            )
        
        # Пагинация
        total = len(items)
        start = (params.page - 1) * params.size
        end = start + params.size
        paginated = items[start:end]
        
        return Page(
            items=[self._to_response(item) for item in paginated],
            page=params.page,
            size=params.size,
            total=total,
            total_pages=(total + params.size - 1) // params.size
        )
    
    def find_by_status_paginated(self, status: str, params: PaginationParams) -> Page[VehicleResponse]:
        items = [v for v in self._vehicles.values() if v['status'] == status]
        
        # Сортировка
        if params.sort_by and params.sort_by in items[0].keys():
            items.sort(
                key=lambda x: x[params.sort_by],
                reverse=params.sort_desc
            )
        
        # Пагинация
        total = len(items)
        start = (params.page - 1) * params.size
        end = start + params.size
        paginated = items[start:end]
        
        return Page(
            items=[self._to_response(item) for item in paginated],
            page=params.page,
            size=params.size,
            total=total,
            total_pages=(total + params.size - 1) // params.size
        )

# Сервисный слой
class AutoparkService:
    def get_vehicles(self, page: int = 1, size: int = 20, 
                     status: Optional[str] = None,
                     sort_by: str = "id", 
                     sort_desc: bool = False) -> Page[VehicleResponse]:
        """Получение ТС с пагинацией и фильтрацией"""
        
        params = PaginationParams(
            page=page,
            size=size,
            sort_by=sort_by,
            sort_desc=sort_desc
        )
        
        if status:
            return self._vehicles.find_by_status_paginated(status, params)
        else:
            return self._vehicles.get_all_paginated(params)
    
    def get_vehicles_due_for_maintenance_paginated(self, page: int = 1, size: int = 20) -> Page[VehicleResponse]:
        """Получение ТС, требующих ТО, с пагинацией"""
        
        # Получаем все ТС, требующие ТО
        due_vehicles = self.get_vehicles_due_for_maintenance()
        
        # Пагинация вручную (или через репозиторий)
        total = len(due_vehicles)
        start = (page - 1) * size
        end = start + size
        paginated = due_vehicles[start:end]
        
        return Page(
            items=paginated,
            page=page,
            size=size,
            total=total,
            total_pages=(total + size - 1) // size
        )
